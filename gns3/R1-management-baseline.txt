! R1 - out-of-band management baseline (Cisco IOSv)
! This represents the "independent SSH management access" the instructor provides.
! Paste ONCE while building your practice lab. It does NOT configure the data plane
! (Gi0/0, Gi0/1, OSPF) - that is done by the automation system.
! Replace <ROUTER_PASSWORD> with the same value you put in .env (ROUTER_SSH_PASSWORD).
enable
configure terminal
hostname R1
no ip domain-lookup
ip domain-name lab.local
username netadmin privilege 15 secret <ROUTER_PASSWORD>
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
interface GigabitEthernet0/3
 description OOB-MGMT - do not modify
 ip address 192.168.100.11 255.255.255.0
 no shutdown
 exit
line vty 0 4
 login local
 transport input ssh
 exec-timeout 30 0
 exit
end
write memory
