! R2 - management access only (Cisco 7200 in GNS3). This is the instructor's part.
! Paste once into the R2 console. First replace <ROUTER_PASSWORD> with ROUTER_SSH_PASSWORD from .env.
enable
configure terminal
hostname R2
no ip domain-lookup
ip domain-name lab.local
username netadmin privilege 15 secret <ROUTER_PASSWORD>
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
interface FastEthernet1/1
 description OOB-MGMT - do not modify
 ip address 192.168.192.12 255.255.255.0
 no shutdown
 exit
line vty 0 4
 login local
 transport input ssh
 exec-timeout 30 0
 exit
end
write memory
